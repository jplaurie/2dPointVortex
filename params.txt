
# Point-vortex example (select the geometry with boundaryCondition)
N 400
# Initial condition choices are random, ring, single, dipole, and file.
# N controls random/ring; single and dipole always create one and two vortices.
# Built-in conditions are generated inside the selected geometry. File input is
# also checked against that geometry, including periodic neutrality.
initialCondition random
# For initialCondition file, provide a whitespace- or comma-separated file with
# x, y, circulation per line. Leave this unset for every built-in condition.
# initialCondition file
# initialConditionFile initial.csv
timeStep 0.001
endTime 100.0
outputTime 1.0
# Optional independent intervals; when omitted each follows outputTime.
# diagnosticsTime 0.005
# checkpointTime 0.1
# Core regularization is supported only for infinite; all other geometries require zero.
coreRadius 0.0
integrator dopri5
absoluteTolerance 1e-10
relativeTolerance 1e-8
minimumTimeStep 1e-12
maximumTimeStep 0.01
numThreads 0
boundaryCondition periodic
# Geometry choices are infinite, periodic_x (periodic in x and unbounded in y),
# periodic (a square torus), and disk. periodic_x uses only boxLengthX. periodic
# uses both box lengths, requires them to be equal, and uses periodicImageLayers.
boxLengthX 1.0
boxLengthY 1.0
periodicImageLayers 8
randomSeed 1234567
# diskRadius is used only by the disk geometry.
diskRadius 1.0
# Remove opposite-sign pairs on the schedule selected below. Reinjection modes
# are none, independent, or paired. Removal uses the minimum x image for periodic_x.
# Reinjection is available only for the bounded periodic and disk domains.
dipoleRemoval true
# Full dipole separation threshold. In a disk this is the real/image distance;
# the image is implicit and is not written as a second vortex.
dipoleRemovalDistance 0.02
# Optionally remove closest-first matched dipoles whose separation is larger
# than this upper distance. Reinjection, when enabled, applies to them as well.
dipoleRemovalUpper false
dipoleRemovalUpperDistance 0.5
# Zero checks after every accepted timestep. A positive value checks at that
# physical-time interval, independently of outputTime; the solver lands exactly on it.
dipoleRemovalInterval 0.0
dipoleReinjection paired
# Every solver run is self-contained. This is the default location; choose a new name for
# each experiment. PointVortex writes trajectory.csv, diagnostics.csv, checkpoints/,
# resolved_parameters.txt, and segments/ here.
runDirectory runs/periodic_n400
# Restarting restores time, timestep, step count, output index, and vortex state.
# Branch a restart into a new run directory.
# restartFile runs/periodic_n400/checkpoints/checkpoint_00000005.dat
# Set true only to replace an existing run's managed solver output.
overwriteRun false
